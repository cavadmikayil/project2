pip install ntc-templates

raw = open("show_cdp_neighbors_detail.txt").read()
neighbors = parse_output(platform="cisco_ios", command="show cdp neighbors detail", data=raw)
for n in neighbors:
    print(n["destination_host"], n["management_ip"], n["local_port"], "→", n["remote_port"])

5. Real Ssenari: MAC Ünvanı Hansı Portdadır?

def find_mac(switches, mac):
    mac = mac.lower().replace(":", "").replace("-", "").replace(".", "")
    mac = f"{mac[0:4]}.{mac[4:8]}.{mac[8:12]}"          # Cisco formatı: aabb.ccdd.eeff
    for sw in switches:
        with ConnectHandler(**sw) as conn:
            table = conn.send_command(f"show mac address-table address {mac}", use_textfsm=True)
            for entry in table if isinstance(table, list) else []:
                ports = entry["destination_port"]
                # Trunk/uplink portları deyil, access portu axtarırıq
                if not any(p.startswith(("Po", "Te", "Twe")) for p in ports):
                    return sw["host"], ports, entry["vlan_id"]
    return None
