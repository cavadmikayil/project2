check_service() {
    local svc="$1"                           # local — funksiya daxili dəyişən
    if systemctl is-active --quiet "$svc"; then
        log INFO "$svc işləyir"
        return 0
    fi
    log WARN "$svc dayanıb"
    return 1
}

{{i:hard-drive-download}} 5. Real Skript: Rotasiyalı Backup

#!/usr/bin/env bash
set -euo pipefail
SRC=/srv/data
DEST=/backup
KEEP_DAYS=14
STAMP=$(date +%F_%H%M)
ARCHIVE="$DEST/data_$STAMP.tar.gz"

mkdir -p "$DEST"
tar -czf "$ARCHIVE" -C "$(dirname "$SRC")" "$(basename "$SRC")"
sha256sum "$ARCHIVE" > "$ARCHIVE.sha256"

# Arxivin oxunduğunu yoxla
tar -tzf "$ARCHIVE" >/dev/null || { echo "Arxiv zədəlidir" >&2; exit 1; }

find "$DEST" -name 'data_*.tar.gz*' -mtime +"$KEEP_DAYS" -delete
echo "OK: $ARCHIVE ($(du -h "$ARCHIVE" | cut -f1))"

Bu skripti əvvəlki dərsdəki systemd timer ilə hər gecə işlədin. Bash 50–100 sətirdən böyüyəndə Python-a keçməyi düşünün.
