{{i:plug}} 1. Qoşulma

from netmiko import ConnectHandler
from getpass import getpass

r1 = {
    "device_type": "cisco_ios",        # cisco_xe, cisco_nxos, juniper_junos, fortinet, mikrotik_routeros...
    "host": "10.10.0.254",
    "username": "netadmin",
    "password": getpass(),
    "secret": getpass("Enable: "),     # enable parolu (lazımdırsa)
}

with ConnectHandler(**r1) as conn:     # blok bitəndə bağlantı avtomatik bağlanır
    conn.enable()
    print(conn.find_prompt())          # R1#

Cihaz tərəfində SSH aktiv olmalıdır: ip domain-name, crypto key generate rsa modulus 2048, line vty 0 4 → transport input ssh, login local.

{{i:eye}} 2. Show Əmrləri

with ConnectHandler(**r1) as conn:
    out = conn.send_command("show ip interface brief")
    print(out)

    # TextFSM ilə strukturlaşdırılmış nəticə (ntc-templates avtomatik gəlir)
    intfs = conn.send_command("show ip interface brief", use_textfsm=True)
    for i in intfs:
        if i["status"] != "up":
            print(f"{i['interface']} → {i['status']}")

    # Uzun çəkən əmrlər üçün
    conn.send_command("show tech-support", read_timeout=300)

use_textfsm=True mətn yerinə dictionary-lərin siyahısını qaytarır — açar adları şablona görə dəyişir (interface, ip_address, status, proto).

{{i:network}} 4. Çoxlu Cihaz: Backup Skripti

import os, yaml
from datetime import date
from concurrent.futures import ThreadPoolExecutor
from netmiko import ConnectHandler, NetmikoTimeoutException, NetmikoAuthenticationException

PASS = os.environ["NET_PASS"]
os.makedirs("backups", exist_ok=True)

def backup(dev):
    params = {"device_type": dev.get("device_type", "cisco_ios"), "host": dev["host"],
              "username": "netadmin", "password": PASS}
    try:
        with ConnectHandler(**params) as conn:
            cfg = conn.send_command("show running-config")
        with open(f"backups/{dev['name']}_{date.today()}.cfg", "w") as f:
            f.write(cfg)
        return dev["name"], "OK"
    except NetmikoTimeoutException:
        return dev["name"], "TIMEOUT"
    except NetmikoAuthenticationException:
        return dev["name"], "AUTH FAIL"

devices = yaml.safe_load(open("inventory.yaml"))["devices"]
with ThreadPoolExecutor(max_workers=10) as pool:           # 10 cihaz paralel
    for name, status in pool.map(backup, devices):
        print(f"{name:15} {status}")

Paralel icra 200 cihazın backup-ını 30 dəqiqədən 2–3 dəqiqəyə endirir. Backup qovluğunu git repo edib hər gecə commit etsəniz, git diff kimin nəyi dəyişdiyini göstərəcək.
